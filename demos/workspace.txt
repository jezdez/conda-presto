
Discover two environments and named Linux targets

$ cat conda.toml
[workspace]
name = "presto-demo"
channels = ["conda-forge"]
channel-priority = "strict"
platforms = [
    { name = "cpu", platform = "linux-64", libc = "2.28" },
    { name = "gpu", platform = "linux-64", libc = "2.28", cuda = "12" },
]

[dependencies]
zlib = ">=1.3,<2"

[feature.tools.dependencies]
zstd = ">=1.5,<2"

[environments]
default = []
tools = ["tools"]

$ conda presto --parse -f conda.toml > discovery.json

$ jq '{environments, selected}' discovery.json
{
  "environments": [
    {
      "name": "default",
      "features": [],
      "no_default_feature": false,
      "platforms": {
        "cpu": "linux-64",
        "gpu": "linux-64"
      }
    },
    {
      "name": "tools",
      "features": [
        "tools"
      ],
      "no_default_feature": false,
      "platforms": {
        "cpu": "linux-64",
        "gpu": "linux-64"
      }
    }
  ],
  "selected": []
}

Select tools/gpu with declared glibc and CUDA versions

$ conda presto --parse -f conda.toml -e tools -p gpu > selected.json

$ jq '.selected[] | {environment, platform, subdir, specs, system_requirements}' selected.json
{
  "environment": "tools",
  "platform": "gpu",
  "subdir": "linux-64",
  "specs": [
    "zlib[version='>=1.3,<2']",
    "zstd[version='>=1.5,<2']"
  ],
  "system_requirements": {
    "glibc": "2.28",
    "cuda": "12"
  }
}

Solve and validate all four environment/target combinations

$ conda presto -f conda.toml --format conda-workspaces-lock-v1 > conda.lock

$ conda presto --validate -f conda.lock --manifest conda.toml > consistency.json

$ jq '{consistent, targets: [.targets[] | {environment, platform, consistent}]}' consistency.json
{
  "consistent": true,
  "targets": [
    {
      "environment": "default",
      "platform": "cpu",
      "consistent": true
    },
    {
      "environment": "default",
      "platform": "gpu",
      "consistent": true
    },
    {
      "environment": "tools",
      "platform": "cpu",
      "consistent": true
    },
    {
      "environment": "tools",
      "platform": "gpu",
      "consistent": true
    }
  ]
}

Reject a manifest that no longer matches the saved lock

$ conda presto --validate -f conda.lock --manifest changed/conda.toml > mismatch.json

$ jq '{consistent, reason: .targets[0].reason}' mismatch.json
{
  "consistent": false,
  "reason": "Dependency 'zlib' in environment 'default' requires 'zlib[version='>=99']' but locked package 'conda-forge/linux-64::zlib-1.3.2-h25fd6f3_3' on 'cpu' does not satisfy it"
}
